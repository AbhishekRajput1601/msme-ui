'DTICSelfEmploymentController',
		function($scope, $loading, $rootScope, $window, $compile,
			$filter, $routeParams, $http, Idle, Keepalive,
			$uibModal) {
			$scope.started = false;

			function closeModals() {
				if ($scope.warning) {
					$scope.warning.close();
					$scope.warning = null;
				}

				if ($scope.timedout) {
					$scope.timedout.close();
					$scope.timedout = null;
				}
			}

			$scope.$on('IdleStart', function() {
				closeModals();

				
				
				
				
			});

			$scope.$on('IdleEnd', function() {
				closeModals();
			});

			$scope
				.$on(
					'IdleTimeout',
					function() {
						closeModals();
						var language = $('#language').val();
						if (language == 'en_US') {
							alert("Your Session has expired, Please relogin.");
						} else if (language == 'hi_IN') {
							alert("आपका सत्र समाप्त हो गया है, कृपया पुनः प्रवेश करें।");
						}
						$window.location.reload();
						
						
						
						
					});
			;

			;

			;

			;

			;

			;

			;

			;

			;

			;

			;

			;

			;

			;

			;

			;

		});
