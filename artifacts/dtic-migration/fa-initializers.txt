			$(document).on('click', '.accept-link', function(event) {
				event.preventDefault(); 
				var $this = $(this); 
				var applicationId = $this.data('application-id'); 

				if (!applicationId) {
					alert('Invalid application ID.');
					return;
				}

				
				if ($this.data('processing') === true) {
					return; 
				}

				
				$this.data('processing', true).text('Processing...');

				
				updateStatus(applicationId, $this);
			});


			function updateStatus(applicationId) {
				
				$.ajax({
					url: 'updateStatus', 
					method: 'POST',
					data: {
						applicationId: applicationId,
						status: 'accepted' 
					},
					success: function(response) {
						
						
						if (response.successMessage) {
							alert(response.successMessage); 
						} else {
							alert('Application status updated successfully.');
							$window.location.href = '#dticfapendingapplicantsunit'

						}

						
						
					},
					error: function(xhr, status, error) {
						
						console.error('Error updating status:', error);
						if (xhr.responseJSON && xhr.responseJSON.errorMessage) {
							alert(xhr.responseJSON.errorMessage); 
						} else {
							alert('An error occurred while updating the status. Please try again.');
						}
					}
				});
			}
