'DTICMSMEController',
		function($scope, $sce, $loading, $rootScope, $window, $compile,
			$filter, $routeParams, $http, $timeout, Idle,
			Keepalive, $uibModal) {
			$scope.started = false;

			function closeModals() {
				if ($scope.warning) {
					$scope.warning.close();
					$scope.warning = null;
				}

				if ($scope.timedout) {
					$scope.timedout.close();
					$scope.timedout = null;
				}
			}

			$scope.$on('IdleStart', function() {
				closeModals();

				
				
				
				
			});

			$scope.$on('IdleEnd', function() {
				closeModals();
			});

			$scope
				.$on(
					'IdleTimeout',
					function() {
						closeModals();
						var language = $('#language').val();
						if (language == 'en_US') {
							alert("Your Session has expired, Please relogin.");
						} else if (language == 'hi_IN') {
							alert("आपका सत्र समाप्त हो गया है, कृपया पुनः प्रवेश करें।");
						}
						$window.location.reload();
						
						
						
						
					});

mpindustry.directive('fileModel', ['$parse', function($parse) {
	return {
		restrict: 'A',
		link: function(scope, element, attrs) {
			var model = $parse(attrs.fileModel);
			var modelSetter = model.assign;

			element.bind('change', function() {
				scope.$apply(function() {
					modelSetter(scope, element[0].files[0]);
					console.log("File selected:", element[0].files[0]); 
				});
			});
		}
	};
}]);
