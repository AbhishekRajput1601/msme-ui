, method = RequestMethod.GET)
	public ModelAndView viewEditLandDetailForm(@PathVariable String vacantLandId, HttpServletRequest request) {
		
		
		
		
		
		

		
		
		
		
		
		
		
		ModelAndView modelAndView = null;
		String dticId = idService.fetchVacantLandDtic(Long.valueOf(vacantLandId));
		
		
		User user = getUserDetail();

		String[] params = {};
		if (user.getUsername().equals(dticId)) {
			modelAndView = new ModelAndView("dtic/editVacantLandForm");
			
			return modelAndView;
		} else {
			
			modelAndView = new ModelAndView("/customError");
			modelAndView.addObject("errorMessage",

			return modelAndView;
		}
