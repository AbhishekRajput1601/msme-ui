, method = RequestMethod.GET)
	public ModelAndView viewIndustrialAreaEditForm(@PathVariable String locationId, HttpServletRequest request) {

		ModelAndView modelAndView = null;
		String dticId = idService.fetchDticByLocationId(Long.valueOf(locationId));
		User user = getUserDetail();

		String[] params = {};
		if (user.getUsername().equals(dticId)) {
			modelAndView = new ModelAndView("dtic/editIndustrialArea");
			return modelAndView;
		} else {
			
			modelAndView = new ModelAndView("/customError");
			modelAndView.addObject("errorMessage",
					MPIndustryUtil.getMessage(MPIndustryConstants.RESPONSE_FILE, "common.illegalAccess", params));
			return modelAndView;
		}
		
	}
