, method = RequestMethod.GET)
	public ModelAndView viewAllotmentLetter(@PathVariable String applicantId, @PathVariable String vacantLandId,
			HttpServletRequest request) {

		ModelAndView modelAndView = null;
		String dticId = idService.fetchDticByApplicantId(Long.valueOf(applicantId));
		User user = getUserDetail();

		String[] params = {};

		if (user.getUsername().equals(dticId)) {

			ESignTxnDetails eSignTxnDetails = esignService.findByPrimaryIdAndType(Long.valueOf(applicantId),
					MPIndustryConstants.TYPE_APPLICATION_AO);
			if (eSignTxnDetails != null && eSignTxnDetails.getIsApplicationFormEsigned() != null
					&& eSignTxnDetails.getIsApplicationFormEsigned()) {
				IDApplicantBean idApplicantBean = idService.fetchApplicationDetail(Long.valueOf(applicantId));
				MultipartFile allotmentFile = idService
						.getCommonsMultipartFileFromDisk(eSignTxnDetails.getNewFilePath());
				idApplicantBean.setAllotmentFile(allotmentFile);
				try {
					uploadAllotmentLetter(idApplicantBean, request);
				} catch (Exception e) {
					
					e.printStackTrace();
				}
				String encryptedIdFromUrl = vacantLandId;
				Long decryptedId = null;
				Long vcId = null;
				String decryptedIdString = "";
				try {
					

					String decodedId = java.net.URLDecoder.decode(encryptedIdFromUrl, "UTF-8");
					decryptedIdString = idService.decrypt(decodedId);
					vcId = Long.parseLong(decryptedIdString);
					decryptedId = Long.parseLong(decryptedIdString);
				} catch (Exception e) {
					
				}
				modelAndView = new ModelAndView("dtic/idPendingApplicantList");
				modelAndView.addObject("decryptedIdString", decryptedId);
				modelAndView.addObject("encryptedString", encryptedIdFromUrl);
				return modelAndView;
			} else {
				modelAndView = new ModelAndView("dtic/viewAllotmentLetter");
				return modelAndView;
			}
		} else {
			
			modelAndView = new ModelAndView("/customError");
			modelAndView.addObject("errorMessage",
					MPIndustryUtil.getMessage(MPIndustryConstants.RESPONSE_FILE, "common.illegalAccess", params));
			return modelAndView;
		}
		

	}
