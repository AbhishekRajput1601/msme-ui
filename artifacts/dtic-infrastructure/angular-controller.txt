			$scope.calTotalEmployment = function(dataArray,
				withPrecision) {
				var total = 0;
				angular.forEach(dataArray, function(data) {
					total += Number(data.empNumber || 0);
				})
				if (withPrecision) {
					return parseFloat(total).toFixed(2);
				} else {
					return parseFloat(total).toFixed(0);
				}
			};

			$scope.formatDateToDDMMYYYY = function(date) {
				if (!date) return '';
				var d = new Date(date);
				var day = ('0' + d.getDate()).slice(-2);
				var month = ('0' + (d.getMonth() + 1)).slice(-2);
				var year = d.getFullYear();
				return day + '/' + month + '/' + year;
			};

				$scope.bidStartBeforePublication = false;
				$scope.bidEndBeforeStart = false;

				if (pubDate && startDate) {
					$scope.bidStartBeforePublication = startDate < pubDate;
				}

				if (startDate && endDate) {
					$scope.bidEndBeforeStart = endDate <= startDate;
				}
			}

			$scope.getRejectCount = function() {
				var counter = 0;

					for (var i = 0; i < $scope.idPendingApplicantList.length; i++) {


						if ($scope.idPendingApplicantList[i].status == "Rejected") {
							counter++;
						}

						if ($scope.idPendingApplicantList[i].statusId == "37") {
							$scope.showTender = true;
						}
						if ($scope.idPendingApplicantList[i].statusId == "38") {
							$scope.showCommitteeForm = true;

						}
						else if ($scope.idPendingApplicantList[i].statusId == "39") {

							$scope.showButton = false;
							$scope.showCommitteeForm = false;
							$scope.tenderInitiate = false;

						}
						else if ($scope.idPendingApplicantList[i].statusId == "40") {
							$scope.showButton = false;
							$scope.showCommitteeForm = false;

						}
						else if ($scope.idPendingApplicantList[i].statusId == "41") {
							$scope.showButton = false;
							$scope.tenderInitiate = false;


						}
					}


				}
				return counter;
			};
