			function validateDates() {
				const pubDate = $scope.formData.publicationDate ? new Date($scope.formData.publicationDate) : null;
				const startDate = $scope.formData.bidStartDate ? new Date($scope.formData.bidStartDate) : null;
				const endDate = $scope.formData.bidEndDate ? new Date($scope.formData.bidEndDate) : null;

				$scope.bidStartBeforePublication = false;
				$scope.bidEndBeforeStart = false;

				if (pubDate && startDate) {
					$scope.bidStartBeforePublication = startDate < pubDate;
				}

				if (startDate && endDate) {
					$scope.bidEndBeforeStart = endDate <= startDate;
				}
			}
