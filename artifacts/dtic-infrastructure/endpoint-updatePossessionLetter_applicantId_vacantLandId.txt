, method = RequestMethod.GET)
	public ModelAndView updatePossessionLetter(@PathVariable String applicantId, @PathVariable String vacantLandId) {

		ModelAndView modelAndView = null;

		ESignTxnDetails esignTxnDetails = esignService.findByPrimaryIdAndTypeAndResponseXml(
				Long.parseLong(applicantId), MPIndustryConstants.TYPE_APPLICANT_APPLICATION_POSSESSION);
		if (esignTxnDetails != null) {
			
			IDApplicant applicant = idService.fetchApplicantDetail(Long.parseLong(applicantId));
			
			applicant.setStatus(new IDMasterStatus(MPIndustryConstants.STATUS_ID_POSSESSION_LETTER_ISSUED));

			
			idService.saveOrUpdateApplicant(applicant);
			IDApplicantBean idApplicantBean = idService.fetchApplicationDetail(Long.valueOf(applicantId));
			MultipartFile possessionFile = idService.getCommonsMultipartFileFromDisk(esignTxnDetails.getNewFilePath());
			idApplicantBean.setPossessionFile(possessionFile);
			idApplicantBean.setPossessionNo(idApplicantBean.getVacantLandBean().getPlotNumber() + "_" + "POSSESSION"
					+ idApplicantBean.getApplicantTitle());
			idApplicantBean.setPossessionDate(new SimpleDateFormat("dd/MM/yyyy").format(new Date()));

			try {
				IDApplicant response = idService.uploadPossessionLetter(idApplicantBean,
						applicant.getUser().getUserId());
			} catch (MPIndustryBusinessException e) {
				
				e.printStackTrace();
			}
			return modelAndView = new ModelAndView("dtic/idPendingApplicantList");
		}
		String dticId = idService.fetchDticByApplicantId(Long.valueOf(applicantId));
		User user = getUserDetail();

		String[] params = {};
		if (user.getUsername().equals(dticId)) {
			modelAndView = new ModelAndView("dtic/uploadPossessionLetter");
			return modelAndView;
		} else {
			
			modelAndView = new ModelAndView("/customError");
			modelAndView.addObject("errorMessage",
					MPIndustryUtil.getMessage(MPIndustryConstants.RESPONSE_FILE, "common.illegalAccess", params));
			return modelAndView;
		}
		

	}
